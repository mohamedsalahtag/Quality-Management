// ---- Program.cs --------------------------------------------------------

using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.DataProtection;
using DemoApp.Models;
using DemoApp.Services;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllersWithViews();

// Cookie auth (precondition).
builder.Services.AddAuthentication(CookieAuthenticationDefaults.AuthenticationScheme)
    .AddCookie(opt =>
    {
        opt.LoginPath        = "/Account/Login";
        opt.ExpireTimeSpan   = TimeSpan.FromDays(30);
        opt.SlidingExpiration = true;
        opt.Cookie.Name      = "DemoApp.Auth";
        opt.Cookie.SameSite  = SameSiteMode.Lax;
        opt.Cookie.HttpOnly  = true;
    });

builder.Services.AddAuthorization(opt =>
{
    opt.AddPolicy("AdminOnly",
        p => p.RequireRole(UserRoles.SiteAdmin));
    opt.AddPolicy("ManagerOrAdmin",
        p => p.RequireRole(UserRoles.Manager, UserRoles.SiteAdmin));
});

// Persistent data-protection keys (precondition).
builder.Services.AddDataProtection()
    .PersistKeysToFileSystem(new DirectoryInfo(
        Path.Combine(builder.Environment.ContentRootPath, "keys")))
    .SetApplicationName("DemoApp");

builder.Services.AddAntiforgery(); // precondition

// ===== view-as-pack: DI =====
builder.Services.AddHttpContextAccessor();
builder.Services.AddScoped<IClaimsTransformation, ViewAsClaimsTransformer>();
// ============================

var app = builder.Build();
app.UseStaticFiles();
app.UseRouting();
app.UseAuthentication();
app.UseAuthorization();
app.MapControllerRoute(
    name: "default",
    pattern: "{controller=Home}/{action=Index}/{id?}");
app.Run();


// ---- Models/User.cs (roles enumeration) --------------------------------

namespace DemoApp.Models;

public static class UserRoles
{
    public const string Viewer    = "Viewer";
    public const string Operator  = "Operator";
    public const string Manager   = "Manager";
    public const string SiteAdmin = "SiteAdmin";

    public static readonly string[] All = { Viewer, Operator, Manager, SiteAdmin };
    public static bool IsValid(string role) => Array.IndexOf(All, role) >= 0;
}

namespace DemoApp.Controllers;

using System.Security.Claims;
using Microsoft.AspNetCore.Mvc;
using DemoApp.Models;
using DemoApp.Services;

public class AccountController : Controller
{
    // ... existing Login / Logout / etc. ...

    [HttpPost, ValidateAntiForgeryToken]
    public IActionResult ViewAs(string? role, string? returnUrl = null)
    {
        var realRole = User.FindFirst(ViewAsClaimsTransformer.OriginalRoleClaim)?.Value
                    ?? User.FindFirst(ClaimTypes.Role)?.Value;
        if (realRole != ViewAsClaimsTransformer.ImpersonatorRole) return Forbid();

        if (string.IsNullOrWhiteSpace(role)
            || role == ViewAsClaimsTransformer.ImpersonatorRole
            || !UserRoles.IsValid(role))
        {
            Response.Cookies.Delete(ViewAsClaimsTransformer.CookieName);
        }
        else
        {
            Response.Cookies.Append(ViewAsClaimsTransformer.CookieName, role, new CookieOptions
            {
                HttpOnly = true,
                SameSite = SameSiteMode.Lax,
                Secure   = Request.IsHttps,
                MaxAge   = TimeSpan.FromHours(8)
            });
        }

        if (!string.IsNullOrEmpty(returnUrl) && Url.IsLocalUrl(returnUrl))
            return Redirect(returnUrl);
        return RedirectToAction("Index", "Home");
    }
}
